cmake_minimum_required(VERSION 3.24)
project(LuckyWheel VERSION 2.0.8 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
configure_file(src/version.h.in generated/version.h @ONLY)
configure_file(assets/app.rc.in generated/app.rc @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.txt" "${PROJECT_VERSION}\n")
include_directories(${CMAKE_CURRENT_BINARY_DIR}/generated)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Sql Multimedia Test)
add_library(wheelcore STATIC src/core.cpp src/core.h src/formats.cpp src/formats.h)
target_include_directories(wheelcore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(wheelcore PUBLIC Qt6::Core Qt6::Sql Qt6::Gui)
add_executable(LuckyWheel WIN32 src/main.cpp src/ms_terms.cpp src/ms_terms.h src/window.cpp src/window.h src/translations.h assets/sounds.qrc assets/icons.qrc assets/legal.qrc assets/translations.qrc)
if(WIN32)
  target_sources(LuckyWheel PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated/app.rc)
endif()
target_link_libraries(LuckyWheel PRIVATE wheelcore Qt6::Widgets Qt6::Multimedia)
if(MSVC)
  target_compile_options(wheelcore PRIVATE /W4 /utf-8)
  target_compile_options(LuckyWheel PRIVATE /W4 /utf-8)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE wheelcore Qt6::Test)
add_test(NAME core_tests COMMAND core_tests -o ${CMAKE_CURRENT_BINARY_DIR}/core-results.txt,txt)
set_tests_properties(core_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(ui_tests tests/ui_tests.cpp src/window.cpp src/window.h src/translations.h assets/sounds.qrc assets/icons.qrc assets/legal.qrc assets/translations.qrc)
target_link_libraries(ui_tests PRIVATE wheelcore Qt6::Widgets Qt6::Multimedia Qt6::Test)
if(MSVC)
  target_compile_options(core_tests PRIVATE /utf-8)
  target_compile_options(ui_tests PRIVATE /utf-8)
endif()
add_test(NAME ui_tests COMMAND ui_tests -o ${CMAKE_CURRENT_BINARY_DIR}/ui-results.txt,txt)
set_tests_properties(ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

# Targeted regression for the Microsoft component acceptance gate.
add_executable(terms_tests tests/terms_tests.cpp src/ms_terms.cpp assets/legal.qrc)
target_include_directories(terms_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(terms_tests PRIVATE Qt6::Widgets Qt6::Test)
if(MSVC)
  target_compile_options(terms_tests PRIVATE /utf-8)
endif()
add_test(NAME terms_tests COMMAND terms_tests)
set_tests_properties(terms_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
